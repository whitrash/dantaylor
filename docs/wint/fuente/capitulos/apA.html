APIs para el código de WINT

Desde Python se llaman con ctypes: el paquete pywin32 no expone estas funciones. Los dos fragmentos siguientes son esqueletos escritos a partir de la documentación de Microsoft; no se ejecutaron en un Windows real durante esta investigación.

# Mantener el equipo despierto mientras dura un trabajo (forma simple)
import ctypes
ES_CONTINUOUS, ES_SYSTEM_REQUIRED = 0x80000000, 0x00000001
k32 = ctypes.windll.kernel32
k32.SetThreadExecutionState(ES_CONTINUOUS | ES_SYSTEM_REQUIRED)
try:
    trabajar()
finally:
    k32.SetThreadExecutionState(ES_CONTINUOUS)   # vuelve al comportamiento normal
# Pedido de energía con motivo visible en «powercfg /requests»
import ctypes
from ctypes import wintypes

class _Detallado(ctypes.Structure):
    _fields_ = [("LocalizedReasonModule", wintypes.HMODULE), ("LocalizedReasonId", wintypes.ULONG),
                ("ReasonStringCount", wintypes.ULONG), ("ReasonStrings", ctypes.POINTER(wintypes.LPWSTR))]
class _Motivo(ctypes.Union):
    _fields_ = [("Detailed", _Detallado), ("SimpleReasonString", wintypes.LPWSTR)]
class REASON_CONTEXT(ctypes.Structure):
    _fields_ = [("Version", wintypes.ULONG), ("Flags", wintypes.DWORD), ("Reason", _Motivo)]

SIMPLE_STRING, SISTEMA, EJECUCION = 0x1, 1, 3     # PowerRequestSystemRequired, PowerRequestExecutionRequired
k32 = ctypes.WinDLL("kernel32", use_last_error=True)
k32.PowerCreateRequest.restype = wintypes.HANDLE
ctx = REASON_CONTEXT(0, SIMPLE_STRING)
ctx.Reason.SimpleReasonString = "WINT Luna: cierre"
h = k32.PowerCreateRequest(ctypes.byref(ctx))
try:
    k32.PowerSetRequest(h, SISTEMA)
    cerrar_el_dia()
finally:
    k32.PowerClearRequest(h, SISTEMA)
    k32.CloseHandle(h)

Otras funciones útiles: PowerRegisterSuspendResumeNotification (saber si el despertar fue automático), SetWaitableTimer con fResume en verdadero (despertar desde S3 o S4 desde un proceso residente) y SetSuspendState (suspender o hibernar de forma explícita).
